## Contexto

O portal dashboard (`/api/portal/dashboard/route.ts`) atualmente faz chamadas LIVE a Klaviyo API para cada store:
- `fetchKlaviyoPerformance()` faz 5-8 API calls por store (metric-aggregates, flow reports, campaign reports, lists, segments)
- Para um cliente com 3 stores, sao 15-24 API calls POR page load
- Isso causa timeouts, rate limits, e load time de 10-30 segundos

O cron `sync-reports` JA popula as seguintes tabelas a cada 2h:
- `store_revenue_summary` — revenue agregado por store/periodo (campaign + flow totals)
- `klaviyo_flow_metrics` — metricas individuais por flow (name, recipients, delivered, open_rate, click_rate, conversion_value, etc.)
- `klaviyo_campaign_metrics` — metricas individuais por campanha (name, recipients, delivered, open_rate, click_rate, conversion_value, send_time, etc.)

Essas tabelas contem TODOS os dados que o portal precisa para revenue, campaigns e flows. O portal so precisa LER do banco em vez de chamar a API.

| `campaignsCount` | list API | COUNT from `klaviyo_campaign_metrics` |
| `flowsCount/activeFlows` | list API | COUNT from `klaviyo_flow_metrics` |

## Objetivo

Reescrever a secao Klaviyo do portal dashboard para ler de tabelas cacheadas (DB) em vez de chamar a Klaviyo API live, reduzindo o load time de 10-30s para < 2s e eliminando rate limit issues.

## Arquitetura Tecnica

### Nova funcao: `fetchKlaviyoFromCache`

```typescript
interface CachedKlaviyoData {
  summary: {
    klaviyo_total_revenue: number
    klaviyo_campaign_revenue: number
    klaviyo_flow_revenue: number
    fetched_at: string
    sync_status: string
  }
  campaigns: Array<{
    campaign_id: string
    campaign_name: string
    campaign_status: string
    send_time: string | null
    recipients: number
    delivered: number
    opened: number
    open_rate: number
    clicked: number
    click_rate: number
    conversion_value: number
    bounce_rate: number
    unsubscribe_rate: number
  }>
  flows: Array<{
    flow_id: string
    flow_name: string
    flow_status: string
    delivered: number
    opened: number
    open_rate: number
    clicked: number
    click_rate: number
    conversion_value: number
    bounce_rate: number
    unsubscribe_rate: number
  }>
}

async function fetchKlaviyoFromCache(
  storeId: string,
  period: string,
  supabase: SupabaseClient
): Promise<CachedKlaviyoData | null> {
  // 1. Get revenue summary to find period dates
  const { data: summary } = await supabase
    .from("store_revenue_summary")
    .select("*")
    .eq("store_id", storeId)
    .eq("period_label", period)
    .single()

  if (!summary) return null

  // 2. Get campaign + flow detail in parallel
  const [campaignsResult, flowsResult] = await Promise.all([
    supabase
      .from("klaviyo_campaign_metrics")
      .select("*")
      .eq("store_id", storeId)
      .eq("period_start", summary.period_start)
      .eq("period_end", summary.period_end)
      .eq("campaign_status", "sent")
      .order("conversion_value", { ascending: false }),
    supabase
      .from("klaviyo_flow_metrics")
      .select("*")
      .eq("store_id", storeId)
      .eq("period_start", summary.period_start)
      .eq("period_end", summary.period_end)
      .order("conversion_value", { ascending: false }),
  ])

  return {
    summary,
    campaigns: campaignsResult.data || [],
    flows: flowsResult.data || [],
  }
}
```

### Mapping de Cache para Portal Format

```typescript
function mapCacheToPortalKlaviyo(cached: CachedKlaviyoData) {
  const campaigns = cached.campaigns
  const flows = cached.flows

  const totalCampDelivered = campaigns.reduce((s, c) => s + (c.delivered || 0), 0)
  const totalFlowDelivered = flows.reduce((s, f) => s + (f.delivered || 0), 0)
  const totalDelivered = totalCampDelivered + totalFlowDelivered

  const totalCampOpened = campaigns.reduce((s, c) => s + (c.opened || 0), 0)
  const totalFlowOpened = flows.reduce((s, f) => s + (f.opened || 0), 0)
  const totalOpened = totalCampOpened + totalFlowOpened

  const totalCampClicked = campaigns.reduce((s, c) => s + (c.clicked || 0), 0)
  const totalFlowClicked = flows.reduce((s, f) => s + (f.clicked || 0), 0)
  const totalClicked = totalCampClicked + totalFlowClicked

  const totalRevenue = cached.summary.klaviyo_total_revenue

  return {
    storeRevenue: 0,
    storeOrders: 0,
    recoveryRate: 0,
    totalLeads: 0,       // Story 8.17
    engagedLeads: 0,     // Story 8.17
    engagementRate: 0,   // Story 8.17
    totalRevenue,
    campaignRevenue: cached.summary.klaviyo_campaign_revenue,
    flowRevenue: cached.summary.klaviyo_flow_revenue,
    smsRevenue: 0,
    emailsSent: totalDelivered,
    delivered: totalDelivered,
    opened: totalOpened,
    clicked: totalClicked,
    openRate: totalDelivered > 0 ? (totalOpened / totalDelivered) * 100 : 0,
    clickRate: totalDelivered > 0 ? (totalClicked / totalDelivered) * 100 : 0,
    clickToOpenRate: totalOpened > 0 ? (totalClicked / totalOpened) * 100 : 0,
    conversionRate: 0,
    unsubscribeRate: /* weighted avg */ 0,
    bounceRate: /* weighted avg */ 0,
    bounces: campaigns.reduce((s, c) => s + (c.bounced || 0), 0) + flows.reduce((s, f) => s + (f.bounced || 0), 0),
    campaignsCount: campaigns.length,
    campaignDelivered: totalCampDelivered,
    campaignRevenuePercent: totalRevenue > 0 ? (cached.summary.klaviyo_campaign_revenue / totalRevenue) * 100 : 0,
    flowsCount: flows.length,
    activeFlows: flows.filter(f => f.flow_status === "live").length,
    flowDelivered: totalFlowDelivered,
    flowRevenuePercent: totalRevenue > 0 ? (cached.summary.klaviyo_flow_revenue / totalRevenue) * 100 : 0,
    recentCampaigns: campaigns.slice(0, 10).map(c => ({
      id: c.campaign_id,
      name: c.campaign_name,
      status: "sent",
      sentAt: c.send_time || new Date().toISOString(),
      recipients: c.recipients,
      delivered: c.delivered,
      opened: c.opened,
      clicked: c.clicked,
      revenue: c.conversion_value,
      openRate: c.open_rate,
      clickRate: c.click_rate,
    })),
    topFlows: flows.slice(0, 10).map(f => ({
      id: f.flow_id,
      name: f.flow_name,
      revenue: f.conversion_value,
      delivered: f.delivered,
      openRate: f.open_rate,
      clickRate: f.click_rate,
    })),
  }
}
```
